FROM python:3.10-slim

# Dependency sistem yang biasa dibutuhkan pytgcalls/yt-dlp (ffmpeg, dll)
RUN apt-get update && apt-get install -y --no-install-recommends \
    ffmpeg \
    git \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /app

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY . .

# Folder ini akan menjadi mount point Volume di Railway
# supaya file session (CosaMusicBot.session) persisten antar deploy.
RUN mkdir -p /app/sessions

CMD ["python", "main.py"]
